1""" 

2Conjugate gradients in fp64, fp32 or fp16, on the CPU with NumPy or on a GPU with CuPy. 

3 

4The vectors and the application of the operator are in the chosen format. Inner products accumulate in fp32 

5(fp64 for fp64), as GPU libraries do for fp16: the products of two fp16 numbers are exact in fp32. The operator 

6is given as a function, so that it can be a stencil without an assembled matrix. 

7""" 

8 

9import numpy as np 

10 

11from pySDC.implementations.problem_classes.HeatEquation_ND_FD import heatNd_unforced

13DTYPES = {'fp64': 'float64', 'fp32': 'float32', 'fp16': 'float16'} 

14 

15 

16_GPU_KERNELS = {} 

17# no fused multiply-adds: every product and sum is rounded on its own, as in the unfused code 

18NO_FMA = ('--fmad=false',) 

19

21def _gpu_kernels(): 

22 """Fused kernels for CG on a GPU: the inner product with a wider accumulator, and the two vector updates.""" 

23 if not _GPU_KERNELS: 

24 import cupy as cp 

25 

26 # the products, widened to the accumulator, in one kernel; summed by cp.sum, which uses CUB as the unfused 

27 # code does, so the inner product is the same number. (A ReductionKernel would reduce a whole vector to a 

28 # scalar with a single thread block, which is far slower.) 

29 for name, accumulate, c_type in (('dot32', 'float32', 'float'), ('dot64', 'float64', 'double')): 

30 product = cp.ElementwiseKernel( 

31 'T u, T v', 

32 f'{accumulate} z', 

33 f'z = {c_type}(u) * {c_type}(v)', 

34 f'pysdc_cg_product_{name}', 

35 options=NO_FMA, 

36 ) 

37 _GPU_KERNELS[name] = lambda u, v, product=product: cp.sum(product(u, v)) 

38 # x += alpha p and r -= alpha Ap, each operation rounded to the format, as the unfused code does 

39 _GPU_KERNELS['update_xr'] = cp.ElementwiseKernel( 

40 'T p, T Ap, T alpha', 

41 'T x, T r', 

42 'x = x + T(alpha * p); r = r - T(alpha * Ap)', 

43 'pysdc_cg_update_xr', 

44 options=NO_FMA, 

45 ) 

46 _GPU_KERNELS['update_p'] = cp.ElementwiseKernel( 

47 'T r, T beta', 'T p', 'p = r + T(beta * p)', 'pysdc_cg_update_p', options=NO_FMA 

48 ) 

49 return _GPU_KERNELS

52def cg(apply, b, fmt, rtol, maxiter, xp=np, fused=False): 

53 """ 

54 Solve A x = b for a symmetric positive definite A, starting from zero. 

55 

56 Parameters 

57 ---------- 

58 apply : callable 

59 Computes A x in the format of x. 

60 b : array 

61 The right-hand side, of order one (fp16 stops at 65504). 

62 fmt : str 

63 'fp64', 'fp32' or 'fp16'. 

64 rtol : float 

65 Stop once the recursively updated residual is below rtol times the norm of b. 

66 maxiter : int 

67 Stop after this many iterations at the latest. 

68 xp : module 

69 numpy or cupy. 

70 fused : bool 

71 On a GPU, use fused kernels for the inner products and the vector updates: the same operations with the 

72 same roundings and the same summation, in fewer passes over memory. 

73 

74 Returns 

75 ------- 

76 tuple 

77 The solution (in the format), the number of iterations and the relative residual reached. 

78 """ 

79 dtype = xp.dtype(DTYPES[fmt]) 

80 accumulate = xp.float64 if fmt == 'fp64' else xp.float32 

81 

82 fused = fused and xp is not np 

83 if fused: 

84 kernels = _gpu_kernels() 

85 reduce = kernels['dot64' if fmt == 'fp64' else 'dot32'] 

86 

87 def dot(u, v): 

88 return float(reduce(u, v)) 

89 

90 else: 

91 

92 def dot(u, v): 

93 return float(xp.sum(u.astype(accumulate) * v.astype(accumulate))) 

94 

95 x = xp.zeros_like(b, dtype=dtype) 

96 r = b.astype(dtype) 

97 p = r.copy() 

98 rr = norm_b = dot(r, r) 

99 k = 0 

100 while k < maxiter and rr > rtol**2 * norm_b: 

101 Ap = apply(p) 

102 pAp = dot(p, Ap) 

103 if pAp <= 0: # the search direction has underflowed in the format: no further progress possible 

104 break 

105 alpha = dtype.type(rr / pAp) 

106 if fused: 

107 kernels['update_xr'](p, Ap, alpha, x, r) 

108 else: 

109 x += alpha * p 

110 r -= alpha * Ap 

111 rr, rr_old = dot(r, r), rr 

112 if not 0 < rr < np.inf: # the residual has vanished or overflowed in the format 

113 break 

114 if fused: 

115 kernels['update_p'](r, dtype.type(rr / rr_old), p) 

116 else: 

117 p = r + dtype.type(rr / rr_old) * p 

118 k += 1 

119 return x, k, float(np.sqrt(rr / norm_b))

125def _gpu_stencil(cp): 

126 """The Jacobi-scaled 1D stencil x - c (x[k-1] + x[k+1]), fused, with the summation order of the roll version.""" 

127 if not _STENCIL: 

128 body = """ 

129 int n = x.size(), k = i; 

130 T s = x[k == 0 ? n - 1 : k - 1] + x[k == n - 1 ? 0 : k + 1]; 

131 y = x[k] - T(c * s) 

132 """ 

133 kernel = cp.ElementwiseKernel('raw T x, T c', 'T y', body, 'pysdc_heat_cg_stencil', options=('--fmad=false',)) 

134 _STENCIL['1d'] = lambda x, c: kernel(x, c, cp.empty_like(x)) 

135 return _STENCIL['1d']

138class heat_cg(heatNd_unforced): 

139 """ 

140 The 1D periodic heat equation with second-order differences, whose solves are CG on the stencil. 

141 

142 The system is divided by its constant diagonal first (Jacobi scaling), which keeps fp16 in range, and the 

143 right-hand side is scaled to a maximum of one. CG runs in ``solve_format`` to the relative residual 

144 ``cg_rtol``; the iterations of every solve are appended to ``cg_iterations``. 

145 """ 

146 

147 def __init__(self, solve_format='fp64', cg_rtol=1e-12, cg_maxiter=1000, **kwargs): 

148 super().__init__(**kwargs) 

149 if len(self.nvars) != 1 or self.bc != 'periodic' or self.order != 2: 

150 raise ValueError('heat_cg needs a 1D periodic grid and second-order differences') 

151 self._makeAttributeAndRegister('solve_format', 'cg_rtol', 'cg_maxiter', localVars=locals()) 

152 self.cg_iterations = [] 

153 

154 def solve_system(self, rhs, factor, u0, t): 

155 xp, n = self.xp, self.nvars[0] 

156 a = factor * self.nu * n**2 

157 diagonal = 1 + 2 * a 

158 coupling = xp.dtype(DTYPES[self.solve_format]).type(a / diagonal) 

159 b = rhs.view(xp.ndarray).reshape(-1) / diagonal 

160 peak = float(xp.abs(b).max()) 

161 me = self.dtype_u(self.init) 

162 if peak == 0: 

163 me[:] = 0.0 

164 return me 

165 gpu = xp is not np # fused kernels on a GPU, bitwise the same as the array operations 

166 apply = ( 

167 (lambda x: _gpu_stencil(xp)(x, coupling)) 

168 if gpu 

169 else (lambda x: x - coupling * (xp.roll(x, 1) + xp.roll(x, -1))) 

170 ) 

171 x, iterations, _ = cg(apply, b / peak, self.solve_format, self.cg_rtol, self.cg_maxiter, xp=xp, fused=gpu) 

172 self.cg_iterations.append(iterations) 

173 me[:] = (x.astype(xp.float64) * peak).reshape(me.shape) 

174 return me
