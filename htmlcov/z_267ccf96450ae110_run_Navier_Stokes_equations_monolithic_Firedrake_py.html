15def setup(t0=3.125e-4): 

16 """ 

17 The parameters of the project's original study. Its residual tolerance, 1e-10 in the L2 norm of the function with 

18 the residual's entries as coefficients, is converted to the dual norm the Firedrake data types use: on the cylinder 

19 mesh, the dual norm of a residual is about 2800 times larger, so 1e-10 becomes about 3e-7. 

20 

21 Returns: 

22 description and controller_params dictionaries, and t0 

23 """ 

24 description = { 

25 'problem_class': NavierStokes2DMonolithicFiredrake, 

26 'problem_params': {'nu': 0.001, 'order': 2}, 

27 'sweeper_class': generic_implicit_mass, 

28 'sweeper_params': {'quad_type': 'RADAU-RIGHT', 'num_nodes': 2, 'QI': 'LU'}, 

29 'level_params': {'restol': 3e-7, 'dt': 1 / 100}, 

30 'step_params': {'maxiter': 10}, 

31 } 

32 return description, {'logger_level': 20, 'hook_class': [LogLiftDrag]}, t0 

33 

34 

35def run_simulation(description, controller_params, t0, Tend): 

36 """ 

37 Run the simulation. 

38 

39 Returns: 

40 The problem, the statistics and the solution at `Tend` 

41 """ 

42 controller = controller_nonMPI(num_procs=1, controller_params=controller_params, description=description) 

43 P = controller.MS[0].levels[0].prob 

44 uend, stats = controller.run(u0=P.u_exact(t0), t0=t0, Tend=Tend) 

45 return P, stats, uend
