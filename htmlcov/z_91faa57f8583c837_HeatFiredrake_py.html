10class Heat1DForcedFiredrake(FormFiredrakeIMEX): 

11 r""" 

12 Forced 1D heat equation with Dirichlet boundary values and Firedrake finite elements, IMEX with the diffusion 

13 implicit, in the mass-matrix formulation and given by its weak forms. 

14 

15 The equation 

16 

17 .. math:: 

18 \frac{\partial u}{\partial t} = \nu \frac{\partial^2 u}{\partial x^2} + f 

19 

20 on :math:`[0, 1]` has the exact solution 

21 

22 .. math:: 

23 u(x, t) = \sin(\pi x) \cos(t) + c, \quad \text{or} \quad u(x, t) = \cos(\pi x) \cos(t) + c 

24 

25 with `time_dependent_bc`, whose Dirichlet values are then :math:`\pm \cos(t) + c`. The forcing 

26 :math:`f = u_t - \nu u_{xx}` is derived from the exact solution by UFL and enters the form as an expression. The 

27 weak formulation 

28 

29 .. math:: 

30 \int_\Omega u_t v\,dx = - \nu \int_\Omega \nabla u \nabla v\,dx + \int_\Omega f v\,dx 

31 

32 gives :math:`M u' = -\nu K u + F`, with the diffusion implicit and the forcing explicit. The mass matrix is never 

33 inverted: use the mass-matrix sweepers, such as 

34 :class:`~pySDC.implementations.sweeper_classes.imex_1st_order_mass.imex_1st_order_mass`. Everything else comes from 

35 :class:`~pySDC.implementations.problem_classes.FormFiredrake.FormFiredrakeIMEX`. 

36 

37 Parameters 

38 ---------- 

39 n : int, optional 

40 Number of cells, ignored if `mesh` is given. 

41 nu : float, optional 

42 Diffusion coefficient :math:`\nu`. 

43 c : float, optional 

44 Constant shift :math:`c` of the solution. 

45 order : int, optional 

46 Degree of the Lagrange elements. 

47 time_dependent_bc : bool, optional 

48 Use the solution with time-dependent boundary values. 

49 LHS_cache_size : int, optional 

50 Number of solvers kept, see ``FormFiredrake``. 

51 mesh : firedrake.MeshGeometry, optional 

52 A mesh of :math:`[0, 1]`, for instance from a ``MeshHierarchy`` for several levels. 

53 comm : MPI communicator, optional 

54 Communicator for spatial parallelism, ignored if `mesh` is given. 

55 """ 

56 

57 def __init__(self, n=30, nu=0.1, c=0.0, order=4, time_dependent_bc=False, LHS_cache_size=12, mesh=None, comm=None): 

58 """Initialization, see the class docstring for the arguments.""" 

59 comm = MPI.COMM_WORLD if comm is None else comm 

60 self.mesh = fd.UnitIntervalMesh(n, comm=comm) if mesh is None else mesh 

61 V = fd.FunctionSpace(self.mesh, 'CG', order) 

62 

63 t = fd.Constant(0.0) 

64 x = fd.SpatialCoordinate(self.mesh)[0] 

65 shape = fd.cos(fd.pi * x) if time_dependent_bc else fd.sin(fd.pi * x) 

66 u_exact = shape * fd.cos(t) + c 

67 forcing = time_derivative(u_exact, t) - nu * fd.div(fd.grad(u_exact)) 

68 

69 def mass(u, v): 

70 return u * v * fd.dx 

71 

72 def diffusion(u, v): 

73 return -nu * fd.inner(fd.grad(u), fd.grad(v)) * fd.dx 

74 

75 def source(u, v): 

76 return forcing * v * fd.dx 

77 

78 super().__init__( 

79 V, 

80 mass, 

81 diffusion, 

82 source, 

83 bcs=(fd.DirichletBC(V, u_exact, 'on_boundary'),), 

84 t=t, 

85 u_exact=u_exact, 

86 LHS_cache_size=LHS_cache_size, 

87 ) 

88 self._makeAttributeAndRegister( 

89 'n', 'order', 'nu', 'c', 'time_dependent_bc', 'comm', localVars=locals(), readOnly=True 

90 ) 

91

95 Spatial coordinate of the mesh, cached after the first access. 

96
