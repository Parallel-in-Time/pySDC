12class NavierStokes2DProjectionFiredrake(FormFiredrakeIMEX): 

13 r""" 

14 Incompressible Navier-Stokes flow around a cylinder (the DFG benchmark 2D-3) with a projection method, IMEX, with 

15 Firedrake, with the standard 

16 :class:`~pySDC.implementations.sweeper_classes.imex_1st_order_mass.imex_1st_order_mass` sweeper and the 

17 convergence controller :class:`~pySDC.projects.StroemungsRaum.convergence_controllers.prepare_step.PrepareStep`, 

18 which tells it where each step starts. 

19 

20 The problem is posed for the velocity :math:`u` alone, with the viscous term implicit and the convection explicit, 

21 

22 .. math:: 

23 \int u_t \cdot v\,dx = -\nu \int \nabla u : \nabla v\,dx - \int (u \cdot \nabla u) \cdot v\,dx, 

24 

25 which is what ``FormFiredrakeIMEX`` gets as forms. Incompressibility enters through the solves: after the implicit 

26 solve for a tentative velocity :math:`u^*` (with the no-slip and inflow conditions at the node's time), the sweeper 

27 passes the step :math:`\Delta\tau` from the start of the step to the node, and 

28 

29 .. math:: 

30 \int \nabla p \cdot \nabla q\,dx = -\frac{1}{\Delta\tau} \int (\nabla \cdot u^*)\, q\,dx, \quad 

31 \int u \cdot v\,dx = \int u^* \cdot v\,dx - \Delta\tau \int \nabla p \cdot v\,dx 

32 

33 project it onto (approximately) divergence-free fields, with :math:`p = 0` at the outflow and the velocity 

34 conditions again in the second solve. The pressure is kept in ``pn``. 

35 

36 Parameters 

37 ---------- 

38 nu : float, optional 

39 Kinematic viscosity :math:`\nu`. 

40 order : int, optional 

41 Degree of the velocity elements; the pressure is one degree lower. 

42 mesh : firedrake.MeshGeometry, optional 

43 The mesh, by default :func:`cylinder_mesh`. 

44 LHS_cache_size : int, optional 

45 Number of solvers for the tentative velocity kept, see ``FormFiredrake``. 

46 """ 

47 

48 def __init__(self, nu=0.001, order=2, mesh=None, LHS_cache_size=12): 

49 """Initialization, see the class docstring for the arguments.""" 

50 mesh = cylinder_mesh() if mesh is None else mesh 

51 V = fd.VectorFunctionSpace(mesh, 'CG', order) 

52 Q = fd.FunctionSpace(mesh, 'CG', order - 1) 

53 

54 def mass(u, v): 

55 return fd.inner(u, v) * fd.dx 

56 

57 def viscous(u, v): 

58 return -nu * fd.inner(fd.nabla_grad(u), fd.nabla_grad(v)) * fd.dx 

59 

60 def convection(u, v): 

61 return -fd.inner(fd.dot(u, fd.nabla_grad(u)), v) * fd.dx 

62 

63 t = fd.Constant(0.0) 

64 super().__init__( 

65 V, 

66 mass, 

67 viscous, 

68 convection, 

69 bcs=velocity_bcs(V, t), 

70 t=t, 

71 u_exact=fd.as_vector([0.0, 0.0]), 

72 LHS_cache_size=LHS_cache_size, 

73 ) 

74 self._makeAttributeAndRegister('nu', 'order', localVars=locals(), readOnly=True) 

75 self.mesh = mesh 

76 

77 # pressure Poisson equation and velocity correction, set up once: the step and u* come in through buffers 

78 self.pn = fd.Function(Q) 

79 #: the start of the current step, set by :meth:`prepare_step` 

80 self._t0 = None 

81 self._dtau = fd.Constant(1.0) 

82 self._u_star = fd.Function(V) 

83 self._u_new = fd.Function(V) 

84 p, q = fd.TrialFunction(Q), fd.TestFunction(Q) 

85 poisson = fd.LinearVariationalProblem( 

86 fd.inner(fd.grad(p), fd.grad(q)) * fd.dx, 

87 -(1 / self._dtau) * fd.div(self._u_star) * q * fd.dx, 

88 self.pn, 

89 bcs=fd.DirichletBC(Q, 0.0, OUTFLOW), 

90 constant_jacobian=True, 

91 ) 

92 u, v = fd.TrialFunction(V), fd.TestFunction(V) 

93 correction = fd.LinearVariationalProblem( 

94 fd.inner(u, v) * fd.dx, 

95 fd.inner(self._u_star, v) * fd.dx - self._dtau * fd.inner(fd.grad(self.pn), v) * fd.dx, 

96 self._u_new, 

97 # the same boundary values as the tentative velocity: the data, or restricted values on coarse levels 

98 bcs=self._solve_bcs, 

99 constant_jacobian=True, 

100 ) 

101 self._poisson = fd.LinearVariationalSolver(poisson) 

102 self._correction = fd.LinearVariationalSolver(correction) 

103 

104 def prepare_step(self, t0, dt, coll): 

105 """ 

106 Remember the start of the step: the projection uses the time from there to the node, see the class docstring. 

107 Called once per step by :class:`~pySDC.projects.StroemungsRaum.convergence_controllers.prepare_step.PrepareStep`. 

108 

109 Args: 

110 t0 (float): Start of the step 

111 dt (float): Step size 

112 coll: The sweeper's collocation rule 

113 """ 

114 self._t0 = t0 

115 

116 def solve_system(self, rhs, factor, u0, t): 

117 r""" 

118 Solve for the tentative velocity, :math:`M u^* - \text{factor}\, R_I(u^*) = \text{rhs}`, and project it with the 

119 step :math:`\Delta\tau = t - t_0` from the start of the step, see the class docstring. The pressure is kept in 

120 ``pn``. 

121 

122 Args: 

123 rhs (firedrake_dual_mesh): Right-hand side, a load vector 

124 factor (float): Prefactor of the implicit right-hand side 

125 u0 (dtype_u): Initial guess 

126 t (float): Time of the node 

127 

128 Returns: 

129 dtype_u: The velocity 

130 """ 

131 if self._t0 is None: 

132 raise RuntimeError('The projection needs the start of the step: add the PrepareStep convergence controller') 

133 u = super().solve_system(rhs, factor, u0, t) 

134 self._u_star.assign(u.functionspace) 

135 self._dtau.assign(t - self._t0) 

136 self._poisson.solve() 

137 self._correction.solve() 

138 u.functionspace.assign(self._u_new) 

139 return u 

140 

141 def drag_lift(self, u): 

142 """ 

143 Drag and lift coefficients of the cylinder with the pressure of the last solve, see :func:`drag_lift`. 

144 

145 Args: 

146 u (dtype_u): Velocity 

147 

148 Returns: 

149 tuple of float: Drag and lift coefficients 

150 """ 

151 return drag_lift(u.functionspace, self.pn, self.nu, self.mesh)
