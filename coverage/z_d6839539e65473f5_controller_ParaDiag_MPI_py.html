11class controller_ParaDiag_MPI(ParaDiagController): 

12 """ 

13 ParaDiag controller with MPI parallelism across time steps: one step per rank. 

14 

15 Everything here is written from a single processor's point of view. A rank owns exactly one step 

16 and never inspects another rank's data; the places where ParaDiag genuinely needs information 

17 from the whole block are expressed as communication: 

18 

19 - ``prepare_Jacobians`` -> Allreduce(SUM) over the step communicator 

20 - ``compute_all_at_once_residual`` -> point-to-point exchange with the previous/next rank 

21 - ``apply_matrix`` (the weighted FFT/iFFT in time) -> a ring reduction, see below 

22 - convergence -> allreduce(LAND), because ParaDiag converges collectively 

23 

24 Note that ParaDiag steps can only converge together, so every rank always participates in every 

25 iteration. That is not a policy choice: a rank that stopped early would never enter the 

26 collectives below and the run would hang. 

27 """ 

28

65 def apply_matrix(self, mat, quantity): 

66 """ 

67 Apply a square L x L matrix across the steps, where L is the number of ranks. 

68 

69 Each rank needs ``res_i = sum_j mat[i, j] * me_j`` but only holds ``me_i``. This is done as a 

70 ring reduction: the values circulate once around the communicator and each rank accumulates 

71 its own row as they pass. That keeps the working set at O(M) fields per rank, independent of 

72 L -- an allgather would instead need O(L * M) fields on every rank, which is exactly the 

73 gather this controller must not do. 

74 

75 The ring costs L - 1 rounds. A butterfly would need only log2(L), but only because the matrix 

76 this is called with is a DFT; for a general matrix it needs the O(L * M) gather just ruled 

77 out. That belongs with an FFT-shaped interface rather than this one, and pays off from about 

78 L = 16 upwards. 

79 

80 Args: 

81 mat: square matrix with as many rows as there are ranks 

82 quantity (str): 'residual' or 'increment', the level attribute to transform in place 

83 """ 

84 comm = self.comm 

85 L, rank = comm.size, comm.rank 

86 assert np.allclose(mat.shape, L), f'need a {L}x{L} matrix, got {mat.shape}' 

87 

88 lvl = self.S.levels[0] 

89 M = lvl.sweep.params.num_nodes 

90 prob = lvl.prob 

91 

92 if quantity == 'residual': 

93 me = lvl.residual 

94 elif quantity == 'increment': 

95 me = lvl.increment 

96 else: 

97 raise NotImplementedError(f'Cannot apply matrix to {quantity!r}') 

98 

99 res = [prob.u_init for _ in range(M)] 

100 # all M nodes travel in one contiguous buffer, so the ring costs L - 1 messages rather than 

101 # M * (L - 1). Same volume, M times fewer message latencies. 

102 held = np.array([me[m] for m in range(M)]) 

103 buf = np.empty_like(held) 

104 

105 nxt, prv = (rank + 1) % L, (rank - 1) % L 

106 for k in range(L): 

107 # after k rotations I am holding the value that started on rank (rank - k) % L 

108 src = (rank - k) % L 

109 for m in range(M): 

110 res[m] += mat[rank, src] * held[m] 

111 

112 if k < L - 1: 

113 comm.Sendrecv(held, dest=nxt, sendtag=k, recvbuf=buf, source=prv, recvtag=k) 

114 held, buf = buf, held 

115 

116 for m in range(M): 

117 me[m] = res[m]

119 def prepare_Jacobians(self): 

120 """Average the solution across all steps, for constructing average Jacobians.""" 

121 if not self.params.average_jacobian: 

122 return 

123 

124 lvl = self.S.levels[0] 

125 M = lvl.sweep.coll.num_nodes 

126 

127 u_avg = [] 

128 for m in range(M): 

129 contribution = lvl.prob.dtype_u(lvl.u[m + 1]) 

130 total = lvl.prob.dtype_u(lvl.prob.init, val=0) 

131 self.comm.Allreduce(contribution, total, op=MPI.SUM) 

132 u_avg.append(total / self.n_steps) 

133 

134 lvl.u_avg = u_avg

136 def compute_all_at_once_residual(self): 

137 """ 

138 Compute the residual of the composite collocation problem. 

139 

140 Needs the previous step's end point as this step's initial condition, which is the only 

141 point-to-point communication in a ParaDiag iteration. 

142 """ 

143 S, comm = self.S, self.comm 

144 lvl = S.levels[0] 

145 

146 lvl.sweep.compute_end_point() 

147 

148 for hook in self.hooks: 

149 hook.pre_comm(step=S, level_number=0) 

150 

151 req = None 

152 if not S.status.last: 

153 req = lvl.uend.isend(dest=(comm.rank + 1) % comm.size, tag=S.status.iter, comm=comm) 

154 if not S.status.first: 

155 lvl.u[0].irecv(source=(comm.rank - 1) % comm.size, tag=S.status.iter, comm=comm).Wait() 

156 if req is not None: 

157 req.Wait() 

158 

159 for hook in self.hooks: 

160 hook.post_comm(step=S, level_number=0, add_to_stats=True) 

161 

162 lvl.sweep.compute_residual()

164 def update_G_inv(self, k=0): 

165 """ 

166 Rebuild this rank's G^-1 if alpha changed with the iteration. 

167 

168 Args: 

169 k (int): 0-based ParaDiag iteration index 

170 """ 

171 alpha = self.get_alpha(k) 

172 if alpha == self._G_inv_alpha: 

173 return 

174 self._G_inv_alpha = alpha 

175 self.S.levels[0].sweep.set_G_inv(get_G_inv_matrix(self.comm.rank, self.comm.size, alpha, self.sweeper_params))

177 def update_solution(self): 

178 """Add the increment to get the next iterate. Purely local.""" 

179 lvl = self.S.levels[0] 

180 for m in range(lvl.sweep.coll.num_nodes): 

181 lvl.u[m + 1] += lvl.increment[m]

198 def it_check(self): 

199 """Check for convergence. ParaDiag converges collectively, hence the allreduce.""" 

200 S, comm = self.S, self.comm 

201 

202 if S.status.iter > 0: 

203 for hook in self.hooks: 

204 hook.post_iteration(step=S, level_number=0) 

205 

206 for C in [self.convergence_controllers[i] for i in self.convergence_controller_order]: 

207 C.post_iteration_processing(self, S, comm=comm) 

208 C.convergence_control(self, S, comm=comm) 

209 

210 for C in [self.convergence_controllers[i] for i in self.convergence_controller_order]: 

211 C.post_iteration_processing_block(self, comm=comm) 

212 

213 for hook in self.hooks: 

214 hook.pre_comm(step=S, level_number=0) 

215 S.status.done = comm.allreduce(S.status.done, op=MPI.LAND) 

216 for hook in self.hooks: 

217 hook.post_comm(step=S, level_number=0, add_to_stats=True) 

218 

219 if not S.status.done: 

220 S.status.iter += 1 

221 for hook in self.hooks: 

222 hook.pre_iteration(step=S, level_number=0) 

223 for C in [self.convergence_controllers[i] for i in self.convergence_controller_order]: 

224 C.pre_iteration_processing(self, S, comm=comm) 

225 S.status.stage = 'IT_PARADIAG' 

226 else: 

227 S.levels[0].sweep.compute_end_point() 

228 for hook in self.hooks: 

229 hook.post_step(step=S, level_number=0) 

230 S.status.stage = 'DONE'

232 def it_ParaDiag(self): 

233 """A single ParaDiag iteration, from this rank's point of view.""" 

234 S = self.S 

235 

236 for hook in self.hooks: 

237 hook.pre_sweep(step=S, level_number=0) 

238 

239 # `it_check` has already incremented the counter, so the first sweep is k = 0 

240 k = max(S.status.iter - 1, 0) 

241 self.update_G_inv(k) 

242 

243 self.prepare_Jacobians() 

244 self.compute_all_at_once_residual() 

245 

246 self.FFT_in_time(quantity='residual', k=k) 

247 S.levels[0].sweep.update_nodes() # local solve, embarrassingly parallel 

248 self.iFFT_in_time(quantity='increment', k=k) 

249 

250 self.update_solution() 

251 

252 for hook in self.hooks: 

253 hook.post_sweep(step=S, level_number=0) 

254 

255 S.status.stage = 'IT_CHECK'

257 def ParaDiag(self): 

258 """Dispatch on the current stage. All ranks are always in the same stage.""" 

259 stage = self.S.status.stage 

260 self.logger.debug(stage) 

261 

262 switcher = {'SPREAD': self.spread, 'IT_CHECK': self.it_check, 'IT_PARADIAG': self.it_ParaDiag} 

263 assert stage in switcher, f'Got unexpected stage {stage!r}' 

264 switcher[stage]() 

265 

266 return self.S.status.done

270 def restart_block(self, time, u0): 

271 """Reset this rank's step for a new block.""" 

272 S, comm = self.S, self.comm 

273 

274 S.status.slot = comm.rank 

275 S.reset_step() 

276 S.status.first = comm.rank == 0 

277 S.status.last = comm.rank == comm.size - 1 

278 S.init_step(u0) 

279 S.status.done = False 

280 S.status.prev_done = False 

281 S.status.iter = 0 

282 S.status.stage = 'SPREAD' 

283 S.status.force_done = False 

284 S.status.time_size = comm.size 

285 

286 for lvl in S.levels: 

287 lvl.tag = None 

288 lvl.status.sweep = 1 

289 lvl.status.time = time 

290 

291 for C in [self.convergence_controllers[i] for i in self.convergence_controller_order]: 

292 C.reset_status_variables(self, comm=comm)

294 def run(self, u0, t0, Tend): 

295 """ 

296 Main driver. 

297 

298 ParaDiag always runs whole blocks -- every rank participates in every iteration -- so unlike 

299 the pipelined controllers there is no communicator splitting as steps finish. 

300 """ 

301 comm = self.comm 

302 for hook in self.hooks: 

303 hook.reset_stats() 

304 

305 all_dt = comm.allgather(self.S.dt) 

306 block_dt = sum(all_dt) 

307 time = t0 + sum(all_dt[: comm.rank]) 

308 block_start = t0 

309 

310 if block_start >= Tend - 10 * np.finfo(float).eps: 

311 raise ControllerError('Nothing to do, check t0, dt and Tend!') 

312 

313 if block_start + block_dt > Tend + 10 * np.finfo(float).eps and comm.rank == 0: 

314 self.logger.warning( 

315 'Warning: This controller will solve past your desired end time until the end of its block!' 

316 ) 

317 

318 self.restart_block(time, u0) 

319 uend = u0 

320 

321 for hook in self.hooks: 

322 hook.post_setup(step=None, level_number=None) 

323 for hook in self.hooks: 

324 hook.pre_run(step=self.S, level_number=0) 

325 

326 while block_start < Tend - 10 * np.finfo(float).eps: 

327 while not self.ParaDiag(): 

328 pass 

329 

330 uend = self.S.levels[0].uend.bcast(root=comm.size - 1, comm=comm) 

331 tend = comm.bcast(self.S.time + self.S.dt, root=comm.size - 1) 

332 

333 for C in [self.convergence_controllers[i] for i in self.convergence_controller_order]: 

334 C.post_step_processing(self, self.S, comm=comm) 

335 C.prepare_next_block(self, self.S, self.S.status.time_size, tend, Tend, comm=comm) 

336 

337 block_start = tend 

338 if block_start < Tend - 10 * np.finfo(float).eps: 

339 all_dt = comm.allgather(self.S.dt) 

340 time = block_start + sum(all_dt[: comm.rank]) 

341 self.restart_block(time, uend) 

342 

343 for hook in self.hooks: 

344 hook.post_run(step=self.S, level_number=0) 

345 for C in [self.convergence_controllers[i] for i in self.convergence_controller_order]: 

346 C.post_run_processing(self, self.S, comm=comm) 

347 

348 return uend, self.return_stats()
