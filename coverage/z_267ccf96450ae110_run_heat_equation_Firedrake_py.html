1""" 

2The forced heat equation with

19DEFAULT_PATH = f'{os.path.dirname(__file__)}/data/heat_equation/'

22def setup(t0=0.0): 

23 """ 

24 The parameters of the project's original study. Its residual tolerance, 1e-12 in the L2 norm of the function with 

25 the residual's entries as coefficients, is converted to the dual norm the Firedrake data types use: on this 64 x 64 

26 mesh with P2 elements, the dual norm of a residual is about 1.2e4 times larger, so 1e-12 becomes about 1e-8. 

27 

28 Returns: 

29 description and controller_params dictionaries, and t0 

30 """ 

31 description = { 

32 'problem_class': Heat2DFiredrake, 

33 'problem_params': {'nu': 0.1, 'c_nvars': 64, 'order': 2, 'c': 0.0}, 

34 'sweeper_class': imex_1st_order_mass, 

35 'sweeper_params': {'quad_type': 'RADAU-RIGHT', 'num_nodes': 2}, 

36 'level_params': {'restol': 1e-8, 'dt': 0.2}, 

37 'step_params': {'maxiter': 20}, 

38 } 

39 controller_params = {'logger_level': 20, 'hook_class': LogSolution} 

40 return description, controller_params, t0 

41 

42 

43def run_simulation(description, controller_params, t0, Tend): 

44 """ 

45 Run the simulation. 

46 

47 Returns: 

48 The problem, the statistics and the relative error at `Tend` 

49 """ 

50 controller = controller_nonMPI(num_procs=1, controller_params=controller_params, description=description) 

51 P = controller.MS[0].levels[0].prob 

52 uend, stats = controller.run(u0=P.u_exact(t0), t0=t0, Tend=Tend) 

53 uex = P.u_exact(Tend) 

54 return P, stats, abs(uex - uend) / abs(uex) 

55 

56 

57def run_postprocessing(description, problem, stats, Tend, path=DEFAULT_PATH): 

58 """ 

59 Write the parameters to a JSON file and the numerical and exact solutions of every step, as ``un`` and ``ux`` with 

60 the step number as index, to ``heat_equation_Firedrake_Temperature.h5``. 

61 """ 

62 Path(path).mkdir(parents=True, exist_ok=True) 

63 parameters = {**description['problem_params'], **description['level_params'], 'Tend': Tend} 

64 with open(f'{path}/heat_equation_Firedrake_parameters.json', 'w') as f: 

65 json.dump(parameters, f) 

66 

67 mesh = problem.V.mesh() 

68 with fd.CheckpointFile(f'{path}/heat_equation_Firedrake_Temperature.h5', 'w', comm=mesh.comm) as file: 

69 file.save_mesh(mesh) 

70 for i, (time, un) in enumerate(get_sorted(stats, type='u', sortby='time')): 

71 file.save_function(un.functionspace, idx=i, name='un') 

72 file.save_function(problem.u_exact(time).functionspace, idx=i, name='ux') 

73 

74 

75if __name__ == '__main__': 

76 Tend = 1.0 

77 description, controller_params, t0 = setup() 

78 problem, stats, rel_err = run_simulation(description, controller_params, t0, Tend) 

79 print('The relative error at time ', Tend, 'is ', rel_err) 

80 run_postprocessing(description, problem, stats, Tend)
