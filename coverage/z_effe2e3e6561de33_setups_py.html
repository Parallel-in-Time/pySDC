170def get_description( 

171 example, nlevels=1, family='CG', coarsening='h', order=None, dt=None, restol=None, maxiter=None, nsteps=None 

172): 

173 """ 

174 Build the pySDC description for one of the three examples. 

175 

176 Parameters 

177 ---------- 

178 example : str 

179 One of :data:`EXAMPLES`. 

180 nlevels : int 

181 1 for SDC, 2 or 3 for MLSDC/PFASST. 

182 family : str 

183 One of :data:`FAMILIES`: ``'CG'`` or ``'DG'``. 

184 coarsening : str 

185 One of :data:`COARSENINGS`: ``'h'`` coarsens the mesh at fixed element order, ``'p'`` drops 

186 the element order on the fine mesh. Irrelevant for ``nlevels=1``. 

187 order : int, optional 

188 Element order for h-coarsening, overriding the default. The refinement ladder is shifted to 

189 keep the dof count on every level, so that changing this isolates the *order* of the coarse 

190 space from its size: in 1d a Lagrange space has ``order * cells + 1`` dofs, so halving the 

191 order and doubling the cells leaves the count alone. Only meaningful with ``coarsening='h'``. 

192 dt, restol, maxiter, nsteps : optional 

193 Override the per-example defaults. 

194 

195 Returns 

196 ------- 

197 description : dict 

198 controller_params : dict 

199 t0, Tend : float 

200 """ 

201 if example not in _SETUPS: 

202 raise ValueError(f'unknown example {example!r}, expected one of {EXAMPLES}') 

203 if family not in FAMILIES: 

204 raise ValueError(f'unknown family {family!r}, expected one of {FAMILIES}') 

205 if coarsening not in COARSENINGS: 

206 raise ValueError(f'unknown coarsening {coarsening!r}, expected one of {COARSENINGS}') 

207 if family not in get_families(example): 

208 raise ValueError(f'{example!r} has no {family} problem class, only {get_families(example)}') 

209 if coarsening not in get_coarsenings(example): 

210 raise ValueError(f'{example!r} is only set up for {get_coarsenings(example)}-coarsening') 

211 if not 1 <= nlevels <= 3: 

212 raise ValueError(f'nlevels must be 1, 2 or 3, got {nlevels}') 

213 

214 s = _SETUPS[example] 

215 dt = s['dt'] if dt is None else dt 

216 nsteps = s['nsteps'] if nsteps is None else nsteps 

217 

218 problem_params = dict(s['problem_params'], family=family) 

219 if coarsening == 'h': 

220 # per level: take the coarsest nlevels entries so the fine level stays fixed 

221 refinements = s['refinements'][:nlevels] 

222 element_order = s['orders'][0] if order is None else order 

223 if order is not None: 

224 # keep the dof count: halving the order and doubling the cells per dimension leaves it 

225 shift = round(math.log2(s['orders'][0] / order)) 

226 refinements = [r + shift for r in refinements] 

227 if min(refinements) < 0: 

228 # dolfin just skips the refine loop, silently giving two identical levels 

229 raise ValueError( 

230 f'{example!r} cannot hold its dof count at order {order}: that needs refinements ' 

231 f'{refinements}, and a negative refinement is not a coarsening' 

232 ) 

233 problem_params['refinements'] = refinements 

234 problem_params['order'] = element_order 

235 orders = [element_order] * nlevels 

236 else: 

237 if order is not None: 

238 raise ValueError('the order override only applies to h-coarsening') 

239 # same mesh everywhere, the fine one; the element order is what comes down 

240 refinements = [s['refinements'][0]] * nlevels 

241 problem_params['refinements'] = s['refinements'][0] 

242 problem_params['order'] = s['orders'][:nlevels] 

243 orders = s['orders'][:nlevels] 

244 

245 if family == 'DG': 

246 # Pin the interior penalty to the fine level. The SIPG form depends on the mesh and order it 

247 # is built on, so rediscretising it on a coarse level would change the penalty by h or p^2 -- 

248 # and the penalty is the dominant term, so the coarse operator would stop being the Galerkin 

249 # operator P^T A_F P of the fine one. alpha_l = sigma p_0^2 h_l / h_0 keeps alpha_l / h_l 

250 # equal on every level, which restores A_G = P^T A_F P exactly. 

251 problem_params['penalty'] = [SIGMA * orders[0] ** 2 * 2 ** (refinements[0] - r) for r in refinements] 

252 

253 sweeper_params = dict(s['sweeper_params']) 

254 sweeper_params['num_nodes'] = [s['num_nodes']] * nlevels 

255 

256 description = { 

257 'problem_class': s['dg_problem_class'] if family == 'DG' else s['problem_class'], 

258 'problem_params': problem_params, 

259 'sweeper_class': s['sweeper_class'], 

260 'sweeper_params': sweeper_params, 

261 'level_params': {'restol': s['restol'] if restol is None else restol, 'dt': dt}, 

262 'step_params': {'maxiter': s['maxiter'] if maxiter is None else maxiter}, 

263 } 

264 

265 if nlevels > 1: 

266 description['space_transfer_class'] = mesh_to_mesh_fenics 

267 description['space_transfer_params'] = {} 

268 description['base_transfer_class'] = base_transfer_mass 

269 # prolong_f is not available for the mass formulation: f is a load vector, so it cannot be 

270 # interpolated. base_transfer_mass falls back to prolong, re-evaluating f on the fine level. 

271 description['base_transfer_params'] = {'finter': False} 

272 

273 controller_params = {'logger_level': 30} 

274 

275 return description, controller_params, 0.0, nsteps * dt
